cmake_minimum_required(VERSION 3.22)
project(cadpilot_engine LANGUAGES CXX)
add_library(cadpilot_engine SHARED ffi/cadpilot_engine.cpp)
target_include_directories(cadpilot_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/ffi)
target_compile_features(cadpilot_engine PUBLIC cxx_std_20)

include(CTest)
if(BUILD_TESTING)
  add_executable(cadpilot_engine_abi_smoke tests/abi_smoke.cpp)
  target_link_libraries(cadpilot_engine_abi_smoke PRIVATE cadpilot_engine)
  add_test(NAME cadpilot_engine_abi_smoke COMMAND cadpilot_engine_abi_smoke)
endif()
